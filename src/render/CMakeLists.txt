qt_add_library(ispview_render STATIC
    bayer_render_parameters.h
    bayer_render_parameters.cpp
    image_canvas.h
    image_canvas.cpp
    navigation_thumbnail_overlay.h
    navigation_thumbnail_overlay.cpp
    yuv_render_parameters.h
    yuv_render_parameters.cpp
)

target_include_directories(ispview_render PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_link_libraries(ispview_render
    PUBLIC ispview_core Qt6::Widgets
    PRIVATE Qt6::GuiPrivate Qt6::ShaderTools
)

qt_add_shaders(ispview_render "ispview_image_shaders"
    PREFIX "/shaders"
    BASE "shaders"
    FILES
        shaders/image.vert
        shaders/image.frag
        shaders/yuv.frag
        shaders/yuv_compare.frag
)

# Packed Bayer decoding needs integer bit operations and texelFetch, which are not
# available in the legacy GLSL 100 ES/120 variants generated by default.
qt_add_shaders(ispview_render "ispview_bayer_shaders"
    PREFIX "/shaders"
    BASE "shaders"
    GLSL "300 es,140,330"
    FILES
        shaders/bayer.frag
        shaders/bayer_compare.frag
        shaders/qml_side.frag
)

ispview_enable_warnings(ispview_render)
